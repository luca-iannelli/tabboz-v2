cmake_minimum_required(VERSION 3.20)
project(TabbozSimulator VERSION 2.0.0 LANGUAGES C RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Build with an x64 Windows compiler")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(GAME_SOURCES disco.c eventi.c lavoro.c proteggi.c readkey.c scooter.c scuola.c tabimg.c telefono.c tempo.c tipa.c vestiti.c zarrosim.c)
add_executable(Tabboz WIN32 ${GAME_SOURCES} native/desktop.c native/window_state.c native/splash.c native/selftest.c native/persistence.c native/money.c native/tabboz.rc)
target_include_directories(Tabboz PRIVATE . native)
target_compile_definitions(Tabboz PRIVATE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 TABBOZ_DESKTOP)
target_compile_options(Tabboz PRIVATE /W3 /utf-8 /wd4068 /wd4996)
target_link_libraries(Tabboz PRIVATE comctl32 comdlg32 gdi32 user32 shell32 advapi32 winmm ole32 uxtheme gdiplus shlwapi)
target_link_options(Tabboz PRIVATE /MANIFEST:NO)
enable_testing()
add_executable(money_test tests/money_test.c native/money.c)
target_include_directories(money_test PRIVATE native)
target_compile_options(money_test PRIVATE /UNDEBUG)
add_test(NAME euro_money COMMAND money_test)
install(TARGETS Tabboz RUNTIME DESTINATION .)
install(FILES LICENSE DESTINATION .)

add_test(NAME native_integration COMMAND Tabboz --self-test)
